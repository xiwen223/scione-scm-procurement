### node_index
节点名称	编号	说明
PP创建	1	采购计划状态为待审批
PP审批	2	采购计划状态为待采购
下单超时	3	PP创建时间→PO创建时间大于24小时
PO创建	4	采购单状态为待提交
PO审批	5	采购单状态为待审批
下发供应商	6	采购单状态为待下单
交付超时	7	PO交付耗时大于15天
财审/付款	8	采购单已下单，结算方式为现结且暂无收货单
CR收货	9	已有CR单暂无IB单
IB创建	10	已有IB单但达成率未完成
流程完成	11	累计入库数满足达成率

```sql
WITH po AS (
    SELECT
        po.order_sn,
        poi.relation_purchase_plan,
        poi.sku,
        po.supplier_name,
        po.ware_house_name,
        po.opt_realname,
        po.create_time,
        poi.quantity_real,
        poi.quantity_entry,
        ibs.ib_sn,
        cr.cr_sn,
        ppsa.plan_sn,
        if(po.order_time is not null,TIMESTAMPDIFF(HOUR, po.order_time, COALESCE(last_cr_create_time,last_ib_create_time,now())),0) delivery_hours,
        case when status = 3 then 4
             when status = 121 then 5
             when status = 1 then 6
             when TIMESTAMPDIFF(DAY, po.order_time, COALESCE(last_cr_create_time,last_ib_create_time,now()))>15 then 7
             when po.order_time is not null and po.settlement_method=7 and cr.cr_sn is null then 8
             when cr.cr_sn is not null and ibs.ib_sn is null then 9
             when ibs.ib_sn is not null and poi.quantity_real>poi.quantity_entry then 10
             when ibs.ib_sn is not null and poi.quantity_real<=poi.quantity_entry then 11
            end node_index
    FROM
        lx_purchase_order_item poi
            LEFT JOIN lx_pp_po_sku_ass ppsa on ppsa.order_sn=poi.order_sn and ppsa.sku=poi.sku
            LEFT JOIN lx_purchase_order po ON po.order_sn = poi.order_sn
            LEFT JOIN (
            SELECT
                ib.purchase_order_sn,
                ibi.sku,
                group_concat(DISTINCT ib.order_sn) ib_sn,
                max(ib.create_time) last_ib_create_time
            FROM
                lx_inbound_order ib
                    LEFT JOIN lx_inbound_order_item ibi ON ib.order_sn = ibi.order_sn
            WHERE
                ib.purchase_order_sn IS NOT NULL
            GROUP BY
                ib.purchase_order_sn,
                ibi.sku
        ) ibs ON ibs.purchase_order_sn = po.order_sn AND ibs.sku = poi.sku
            left join(
            SELECT
                cr.business_order_sn,
                cri.sku,
                group_concat(DISTINCT cr.order_sn) cr_sn,
                max(cr.create_time) last_cr_create_time
            FROM
                lx_receipt_order cr
                    LEFT JOIN lx_receipt_order_item cri ON cr.order_sn = cri.order_sn
            GROUP BY
                cr.business_order_sn,
                cri.sku
        ) cr on cr.business_order_sn=poi.order_sn and poi.sku=cr.sku
    WHERE
        poi.is_delete = 0
      AND po.`status` NOT IN (- 1, 124)
)
SELECT
    pp.plan_sn,
    pp.quantity_plan pp_plan_qty,
    pp.create_time pp_create_at,
    pp.status_text,
    GROUP_CONCAT(DISTINCT COALESCE(po.supplier_name,po1.supplier_name)) supplier_names,
    GROUP_CONCAT(DISTINCT COALESCE(po.ware_house_name,po1.ware_house_name)) warehouse_names,
    GROUP_CONCAT(DISTINCT COALESCE(po.opt_realname,po1.opt_realname)) purchasers,
    pp.approval_time pp_audit_at,
    pp.create_time pp_create_time,
    MAX(COALESCE(po.create_time,po1.create_time)) po_created_time,
    GROUP_CONCAT(DISTINCT COALESCE(po.order_sn,po1.order_sn)) order_sns,
    TIMESTAMPDIFF(HOUR, pp.create_time, IF(MAX(po.create_time) IS NOT NULL, MAX(po.create_time), now())) order_risk,
    CASE
        WHEN TIMESTAMPDIFF(HOUR, pp.create_time, IF(MAX(COALESCE(po.create_time,po1.create_time)) IS NOT NULL, MAX(COALESCE(po.create_time,po1.create_time)), now())) > 24 THEN	'超时'
        WHEN TIMESTAMPDIFF(HOUR, pp.create_time, IF(MAX(COALESCE(po.create_time,po1.create_time)) IS NOT NULL, MAX(COALESCE(po.create_time,po1.create_time)), now())) <= 24 AND MAX(COALESCE(po.create_time,po1.create_time)) IS NULL THEN '下单中'
        WHEN TIMESTAMPDIFF(HOUR, pp.create_time, IF(MAX(COALESCE(po.create_time,po1.create_time)) IS NOT NULL, MAX(COALESCE(po.create_time,po1.create_time)), now())) <= 24 AND MAX(COALESCE(po.create_time,po1.create_time)) IS NOT NULL THEN '及时'
        END order_risk_text,
    pp.status_text,
    case
        when pp.status_text='待审批' then 1
        when pp.status_text='待采购' then 2
        when TIMESTAMPDIFF(HOUR, pp.create_time, IF(MAX(COALESCE(po.create_time,po1.create_time)) IS NOT NULL, MAX(COALESCE(po.create_time,po1.create_time)), now())) > 24 then 3
        else max(COALESCE(po.node_index,po1.node_index))
        end node_index
FROM
    lx_purchase_plan pp
        LEFT JOIN po ON po.sku=pp.sku and po.plan_sn=pp.plan_sn
        LEFT JOIN lx_plan_combo_relation cr on cr.plan_sn=pp.plan_sn
        LEFT JOIN po as po1 ON po1.sku=cr.sku and po1.plan_sn=pp.plan_sn
WHERE
    pp.STATUS NOT IN (- 3, 124, 122)
  AND pp.create_time >= '2026-08-01'
GROUP BY
    pp.plan_sn
```
